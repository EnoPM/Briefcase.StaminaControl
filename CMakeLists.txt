cmake_minimum_required(VERSION 3.28)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" MOD_VERSION)
string(STRIP "${MOD_VERSION}" MOD_VERSION)
if(NOT MOD_VERSION MATCHES "^(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)$")
 message(FATAL_ERROR "VERSION must contain a stable MAJOR.MINOR.PATCH version.")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION")
project(Briefcase.StaminaControl VERSION ${MOD_VERSION} LANGUAGES CXX)
configure_file(briefcase.mod.json.in briefcase.mod.json @ONLY)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/mod-build.json" _build_config)
string(JSON _sdk_version GET "${_build_config}" sdkVersion)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
find_package(BriefcaseNativeSDK ${_sdk_version} EXACT CONFIG REQUIRED)
add_library(Briefcase.SuspicionControl SHARED StaminaControlMod.cpp SuspicionControl.cpp)
target_link_libraries(Briefcase.SuspicionControl PRIVATE Briefcase::ModApi Briefcase::Json)
set_target_properties(Briefcase.SuspicionControl PROPERTIES PREFIX "")
include(CTest)
if(BUILD_TESTING)
 add_executable(ModContracts tests/Contracts.cpp)
 target_link_libraries(ModContracts PRIVATE Briefcase::ModApi Briefcase::Json)
 add_test(NAME ModContracts COMMAND ModContracts)
 add_executable(ModLifecycle tests/Lifecycle.cpp)
 target_link_libraries(ModLifecycle PRIVATE Briefcase::ModApi Briefcase::Json ${CMAKE_DL_LIBS})
 target_compile_definitions(ModLifecycle PRIVATE MOD_SOURCE="${CMAKE_CURRENT_SOURCE_DIR}")
 add_dependencies(ModLifecycle Briefcase.SuspicionControl)
 add_test(NAME ModLifecycle COMMAND ModLifecycle)
 set_tests_properties(ModLifecycle PROPERTIES WORKING_DIRECTORY "$<TARGET_FILE_DIR:Briefcase.SuspicionControl>" TIMEOUT 20)
endif()
install(TARGETS Briefcase.SuspicionControl LIBRARY DESTINATION Briefcase/Mods/briefcase.suspicion-control RUNTIME DESTINATION Briefcase/Mods/briefcase.suspicion-control)
set(mod_manifest "${CMAKE_CURRENT_BINARY_DIR}/briefcase.mod.json")
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
 file(READ "${mod_manifest}" manifest)
 string(REPLACE ".dll" ".so" manifest "${manifest}")
 file(WRITE "${mod_manifest}" "${manifest}")
endif()
install(FILES "${mod_manifest}" DESTINATION Briefcase/Mods/briefcase.suspicion-control)
install(FILES Data/config.json DESTINATION Briefcase/Mods/briefcase.suspicion-control/Data)
