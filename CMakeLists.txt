cmake_minimum_required(VERSION 3.28)
project(Briefcase.StaminaControl VERSION 0.3.0 LANGUAGES CXX)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/mod-build.json" _build_config)
string(JSON _sdk_version GET "${_build_config}" sdkVersion)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(BriefcaseNativeSDK ${_sdk_version} EXACT CONFIG REQUIRED)
add_library(Briefcase.SuspicionControl SHARED StaminaControlMod.cpp SuspicionControl.cpp)
target_link_libraries(Briefcase.SuspicionControl PRIVATE Briefcase::ModApi Briefcase::Json)
set_target_properties(Briefcase.SuspicionControl PROPERTIES PREFIX "")
include(CTest)
if(BUILD_TESTING)
 add_executable(ModContracts tests/Contracts.cpp)
 target_link_libraries(ModContracts PRIVATE Briefcase::ModApi Briefcase::Json)
 add_test(NAME ModContracts COMMAND ModContracts)
 add_executable(ModLifecycle tests/Lifecycle.cpp)
 target_link_libraries(ModLifecycle PRIVATE Briefcase::ModApi Briefcase::Json)
 target_compile_definitions(ModLifecycle PRIVATE MOD_SOURCE="${CMAKE_CURRENT_SOURCE_DIR}")
 add_dependencies(ModLifecycle Briefcase.SuspicionControl)
 add_test(NAME ModLifecycle COMMAND ModLifecycle)
 set_tests_properties(ModLifecycle PROPERTIES WORKING_DIRECTORY "$<TARGET_FILE_DIR:Briefcase.SuspicionControl>" TIMEOUT 20)
endif()
install(TARGETS Briefcase.SuspicionControl RUNTIME DESTINATION Briefcase/Mods/briefcase.suspicion-control)
install(FILES briefcase.mod.json DESTINATION Briefcase/Mods/briefcase.suspicion-control)
install(FILES Data/config.json DESTINATION Briefcase/Mods/briefcase.suspicion-control/Data)
